//Implemented a VHDL-based Ping-Pong game on a Basys 3 FPGA , utilizing a Finite State Machine (FSM) to control ball movement across a 16-LED array and a multiplexed 7-segment display for //real-time score tracking of two players


library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.NUMERIC_STD.ALL;

entity pong_game is
    Port (
        clk   : in  STD_LOGIC;
        rst   : in  STD_LOGIC;
        btnL  : in  STD_LOGIC;
        btnR  : in  STD_LOGIC;
        led   : out STD_LOGIC_VECTOR (15 downto 0);
        seg   : out STD_LOGIC_VECTOR (6 downto 0);
        an    : out STD_LOGIC_VECTOR (3 downto 0)
    );
end pong_game;

architecture Behavioral of pong_game is

    
    type state_type is (ST_IDLE, ST_MOVING, ST_SCORE_WAIT);
    signal state : state_type := ST_IDLE;

    type direction_type is (LEFT, RIGHT);
    signal dir : direction_type := LEFT;

    
    signal led_index : integer range 0 to 15 := 15;
    signal led_reg   : std_logic_vector(15 downto 0) := (others => '0');
    signal counter   : integer := 0;
    signal score_p1  : integer range 0 to 99 := 0;
    signal score_p2  : integer range 0 to 99 := 0;

   
    signal slow_clk      : std_logic := '0';
    signal clk_div_count : integer := 0;
    constant CLK_DIV_MAX : integer := 5_000_000;
    
    signal slow_clk1      : std_logic := '0';
    signal clk_div_count1 : integer := 0;
     constant CLK_DIV_MAX1 : integer := 100_000;

   
    signal seg_data        : std_logic_vector(6 downto 0) := (others => '1');
    signal an_data         : std_logic_vector(3 downto 0) := "1111";
    signal refresh_counter : integer range 0 to 3 := 0;

    
    function digit_to_7seg(d : integer range 0 to 9) return std_logic_vector is
    begin
        case d is
            when 0 => return "1000000";
            when 1 => return "1111001";
            when 2 => return "0100100";
            when 3 => return "0110000";
            when 4 => return "0011001";
            when 5 => return "0010010";
            when 6 => return "0000010";
            when 7 => return "1111000";
            when 8 => return "0000000";
            when 9 => return "0010000";
            when others => return "1111111";
        end case;
    end function;

begin
   
    led <= led_reg;
    seg <= seg_data;
    an  <= an_data;

    
    process(clk)
    begin
        if rising_edge(clk) then
            if clk_div_count = CLK_DIV_MAX/2 - 1 then
                clk_div_count <= 0;
                slow_clk <= not slow_clk;
            else
                clk_div_count <= clk_div_count + 1;
            end if;
        end if;
    end process;
    
    process(clk)
    begin
        if rising_edge(clk) then
            if clk_div_count1 = CLK_DIV_MAX1/2 - 1 then
                clk_div_count1 <= 0;
                slow_clk1 <= not slow_clk1;
            else
                clk_div_count1 <= clk_div_count1 + 1;
            end if;
        end if;
    end process;

    
    
    process(slow_clk)
    begin
        if rising_edge(slow_clk) then
            if rst = '1' then
                state <= ST_IDLE;
                dir <= LEFT;
                led_index <= 15;
                led_reg <= (others => '0');
                led_reg(15) <= '1';
                score_p1 <= 0;
                score_p2 <= 0;
                counter <= 0;
            else
                case state is
                    when ST_IDLE =>
                        led_reg <= (others => '0');
                        led_reg(15) <= '1';
                        led_index <= 15;
                        dir <= LEFT;
                        if btnL = '1' then
                            state <= ST_MOVING;
                            led_index <= 14;
                            led_reg <= (others => '0');
                            led_reg(14) <= '1';
                        end if;

                    when ST_MOVING =>
                        led_reg <= (others => '0');
                        if dir = LEFT then
                            if led_index = 0 then
                                if btnR = '1' then
                                    dir <= RIGHT;
                                    led_index <= 1;
                                    led_reg(1) <= '1';
                                else
                                    state <= ST_SCORE_WAIT;
                                    score_p1 <= score_p1 + 1;
                                    led_reg <= (others => '1');
                                end if;
                            else
                                led_index <= led_index - 1;
                                led_reg(led_index - 1) <= '1';
                            end if;
                        else
                            if led_index = 15 then
                                if btnL = '1' then
                                    dir <= LEFT;
                                    led_index <= 14;
                                    led_reg(14) <= '1';
                                else
                                    state <= ST_SCORE_WAIT;
                                    score_p2 <= score_p2 + 1;
                                    led_reg <= (others => '1');
                                end if;
                            else
                                led_index <= led_index + 1;
                                led_reg(led_index + 1) <= '1';
                            end if;
                        end if;

                    when ST_SCORE_WAIT =>
                        if counter < 5 then 
                            counter <= counter + 1;
                        else
                            counter <= 0;
                            state <= ST_IDLE;
                            led_index <= 15;
                            led_reg <= (others => '0');
                            led_reg(15) <= '1';
                        end if;
                end case;
            end if;
        end if;
    end process;

    
    process(slow_clk1)
        variable d0, d1, d2, d3 : integer range 0 to 9;
    begin
        if rising_edge(slow_clk1) then
            
            d0 := score_p2 mod 10; 
            d1 := score_p2 / 10;   
            d2 := score_p1 mod 10; 
            d3 := score_p1 / 10;   

            case refresh_counter is
                when 0 =>
                    seg_data <= digit_to_7seg(d0);
                    an_data <= "1110";
                when 1 =>
                    seg_data <= digit_to_7seg(d1);
                    an_data <= "1101";
                when 2 =>
                    seg_data <= digit_to_7seg(d2);
                    an_data <= "1011";
                when 3 =>
                    seg_data <= digit_to_7seg(d3);
                    an_data <= "0111";
                when others =>
                    null;
            end case;

            refresh_counter <= (refresh_counter + 1) mod 4;
        end if;
    end process;

end Behavioral;
